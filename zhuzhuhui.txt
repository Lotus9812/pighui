    <script>
        // 获取所有输入框元素
        const quantity8 = document.getElementById('quantity8');
        const quantity12 = document.getElementById('quantity12');
        const quantity16 = document.getElementById('quantity16');
        const customPrice = document.getElementById('customPrice');
        
        // 获取所有显示小计的元素
        const subtotal8 = document.getElementById('subtotal8');
        const subtotal12 = document.getElementById('subtotal12');
        const subtotal16 = document.getElementById('subtotal16');
        const subtotalCustom = document.getElementById('subtotalCustom');
        
        // 获取总计相关元素
        const subtotalElement = document.getElementById('subtotal');
        const serviceFeeElement = document.getElementById('serviceFee');
        const totalElement = document.getElementById('total');
        
        // 添加事件监听器
        quantity8.addEventListener('input', calculateTotal);
        quantity12.addEventListener('input', calculateTotal);
        quantity16.addEventListener('input', calculateTotal);
        customPrice.addEventListener('input', calculateTotal);
        
        // 计算总价的函数
        function calculateTotal() {
            // 获取数量值
            const qty8 = parseInt(quantity8.value) || 0;
            const qty12 = parseInt(quantity12.value) || 0;
            const qty16 = parseInt(quantity16.value) || 0;
            const priceCustom = parseFloat(customPrice.value) || 0;
            
            // 计算每种商品的小计
            const st8 = qty8 * 8;
            const st12 = qty12 * 12;
            const st16 = qty16 * 16;
            const stCustom = priceCustom;
            
            // 更新每种商品的小计显示
            subtotal8.textContent = st8.toFixed(2) + '元';
            subtotal12.textContent = st12.toFixed(2) + '元';
            subtotal16.textContent = st16.toFixed(2) + '元';
            subtotalCustom.textContent = stCustom.toFixed(2) + '元';
            
            // 计算商品总价
            const subtotal = st8 + st12 + st16 + stCustom;
            
            // 计算服务费（10%）
            const serviceFee = subtotal * 0.1;
            
            // 计算总计
            const total = subtotal + serviceFee;
            
            // 更新显示
            subtotalElement.textContent = subtotal.toFixed(2) + '元';
            serviceFeeElement.textContent = serviceFee.toFixed(2) + '元';
            totalElement.textContent = total.toFixed(2) + '元';
        }
        
        // 页面加载时初始化计算
        calculateTotal();
    </script>
